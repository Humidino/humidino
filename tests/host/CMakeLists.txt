cmake_minimum_required(VERSION 3.20)
project(humidino_host_checks LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "../.." ABSOLUTE)
set(LVGL "${ROOT}/.pio/libdeps/esp32-s3-devkitc-1/lvgl")
file(GLOB_RECURSE LVGL_SOURCES "${LVGL}/src/*.c")
add_library(lvgl STATIC ${LVGL_SOURCES})
target_include_directories(lvgl PUBLIC "${ROOT}/include" "${LVGL}")
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE LV_LVGL_H_INCLUDE_SIMPLE)
add_executable(firmware_checks main.cpp "${ROOT}/src/settings_actions.cpp"
    "${ROOT}/src/fonts/font_ru_14.c" "${ROOT}/src/fonts/font_ru_20.c")
target_include_directories(firmware_checks PRIVATE stubs "${ROOT}/src")
target_link_libraries(firmware_checks PRIVATE lvgl)
if(MSVC)
    target_compile_options(firmware_checks PRIVATE /utf-8)
endif()
enable_testing()
add_test(NAME firmware_checks COMMAND firmware_checks)
add_executable(backlight_checks backlight.cpp "${ROOT}/src/backlight.cpp")
target_include_directories(backlight_checks PRIVATE stubs "${ROOT}/src" "${ROOT}/include")
add_test(NAME backlight_checks COMMAND backlight_checks)
